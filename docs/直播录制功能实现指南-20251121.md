## 📊 数据库结构

### live_recordings 表

```sql
CREATE TABLE live_recordings (
  id UUID PRIMARY KEY,
  session_id UUID,                    -- 关联的直播会话
  title TEXT,                          -- 录制标题
  description TEXT,                    -- 描述信息
  file_path TEXT,                      -- 存储路径
  file_url TEXT,                       -- 访问URL
  file_size BIGINT,                    -- 文件大小（字节）
  duration_seconds INTEGER,            -- 时长（秒）
  recorded_by UUID,                    -- 录制者
  status TEXT,                         -- processing/completed/failed
  created_at TIMESTAMPTZ,
  updated_at TIMESTAMPTZ
);
```

**字段说明**：
- `status`：
  - `processing`：录制中或处理中
  - `completed`：已完成，可播放
  - `failed`：录制失败

### Storage 桶

**桶名**：`live-recordings`
**访问**：公开（用于回放）
**路径结构**：`{user_id}/{session_id}/{filename}`

**示例**：
```
live-recordings/
├── abc-123-user-id/
│   └── xyz-789-session-id/
│       └── recording-2025-11-21.mp4
```

#### 步骤 4：创建录制 API 路由

创建 `src/app/api/live-sessions/[id]/recording/route.ts`：

```typescript
import { NextRequest, NextResponse } from "next/server";
import { RoomServiceClient, EgressClient } from 'livekit-server-sdk';
import { authenticateRequest } from "@/lib/server/auth";

const roomService = new RoomServiceClient(
  process.env.LIVEKIT_URL!,
  process.env.LIVEKIT_API_KEY!,
  process.env.LIVEKIT_API_SECRET!
);

const egressClient = new EgressClient(
  process.env.LIVEKIT_URL!,
  process.env.LIVEKIT_API_KEY!,
  process.env.LIVEKIT_API_SECRET!
);

// POST - 开始录制
export async function POST(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    const user = await authenticateRequest(req);
    if (!user) {
      return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
    }

    const sessionId = params.id;
    const roomName = `live-session-${sessionId}`;

    // 启动房间录制
    const egress = await egressClient.startRoomCompositeEgress(roomName, {
      fileOutputs: [{
        fileType: 'MP4',
        filepath: `${user.id}/${sessionId}/recording-{time}.mp4`,
      }],
      // 上传到 S3 或其他存储（可选）
      s3: {
        accessKey: process.env.S3_ACCESS_KEY,
        secret: process.env.S3_SECRET,
        bucket: 'live-recordings',
        region: 'us-west-2',
      }
    });

    // 保存录制记录到数据库
    const { data, error } = await supabase
      .from('live_recordings')
      .insert({
        session_id: sessionId,
        title: `录制 - ${new Date().toLocaleString()}`,
        recorded_by: user.id,
        status: 'processing',
        egress_id: egress.egressId, // LiveKit 录制 ID
      })
      .select()
      .single();

    if (error) throw error;

    return NextResponse.json({
      recordingId: data.id,
      egressId: egress.egressId,
    });
  } catch (error: any) {
    console.error('启动录制失败:', error);
    return NextResponse.json(
      { error: error.message },
      { status: 500 }
    );
  }
}

// DELETE - 停止录制
export async function DELETE(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    const user = await authenticateRequest(req);
    if (!user) {
      return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
    }

    const body = await req.json();
    const { egressId } = body;

    // 停止录制
    await egressClient.stopEgress(egressId);

    // 更新数据库状态
    const { error } = await supabase
      .from('live_recordings')
      .update({ status: 'completed' })
      .eq('egress_id', egressId);

    if (error) throw error;

    return NextResponse.json({ success: true });
  } catch (error: any) {
    console.error('停止录制失败:', error);
    return NextResponse.json(
      { error: error.message },
      { status: 500 }
    );
  }
}
```

#### 步骤 6：处理录制完成回调

LiveKit 会在录制完成后发送 Webhook。创建 `src/app/api/webhooks/livekit/route.ts`：

```typescript
import { NextRequest, NextResponse } from "next/server";
import { WebhookReceiver } from 'livekit-server-sdk';

const receiver = new WebhookReceiver(
  process.env.LIVEKIT_API_KEY!,
  process.env.LIVEKIT_API_SECRET!
);

export async function POST(req: NextRequest) {
  try {
    const body = await req.text();
    const signature = req.headers.get('Authorization') || '';

    // 验证 webhook
    const event = receiver.receive(body, signature);

    if (event.event === 'egress_ended') {
      const { egressId, status, fileResults } = event;

      // 更新数据库
      if (fileResults && fileResults.length > 0) {
        const file = fileResults[0];

        await supabase
          .from('live_recordings')
          .update({
            status: status === 'complete' ? 'completed' : 'failed',
            file_url: file.downloadUrl,
            file_size: file.size,
            duration_seconds: file.duration,
          })
          .eq('egress_id', egressId);
      }
    }

    return NextResponse.json({ success: true });
  } catch (error: any) {
    console.error('Webhook 处理失败:', error);
    return NextResponse.json(
      { error: error.message },
      { status: 500 }
    );
  }
}
```

### 方案 3：客户端录制（快速测试方案）

这是一个**简化的实现**，适合开发测试，但有诸多限制。

#### 实现代码

创建 `src/hooks/useMediaRecorder.ts`：

```typescript
import { useState, useRef, useCallback } from 'react';

export function useMediaRecorder() {
  const [isRecording, setIsRecording] = useState(false);
  const [recordedBlob, setRecordedBlob] = useState<Blob | null>(null);
  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
  const chunksRef = useRef<Blob[]>([]);

  const startRecording = useCallback(async () => {
    try {
      // 获取屏幕和音频流
      const displayStream = await navigator.mediaDevices.getDisplayMedia({
        video: { mediaSource: 'screen' },
        audio: true,
      });

      const audioStream = await navigator.mediaDevices.getUserMedia({
        audio: true,
      });

      // 合并流
      const combinedStream = new MediaStream([
        ...displayStream.getVideoTracks(),
        ...audioStream.getAudioTracks(),
      ]);

      // 创建 MediaRecorder
      const mediaRecorder = new MediaRecorder(combinedStream, {
        mimeType: 'video/webm;codecs=vp9',
      });

      chunksRef.current = [];

      mediaRecorder.ondataavailable = (event) => {
        if (event.data.size > 0) {
          chunksRef.current.push(event.data);
        }
      };

      mediaRecorder.onstop = () => {
        const blob = new Blob(chunksRef.current, { type: 'video/webm' });
        setRecordedBlob(blob);
        setIsRecording(false);

        // 停止所有轨道
        combinedStream.getTracks().forEach(track => track.stop());
      };

      mediaRecorder.start(1000); // 每秒保存一次数据
      mediaRecorderRef.current = mediaRecorder;
      setIsRecording(true);
    } catch (error) {
      console.error('启动录制失败:', error);
      throw error;
    }
  }, []);

  const stopRecording = useCallback(() => {
    if (mediaRecorderRef.current && isRecording) {
      mediaRecorderRef.current.stop();
    }
  }, [isRecording]);

  const uploadRecording = useCallback(async (
    sessionId: string,
    userId: string
  ) => {
    if (!recordedBlob) return null;

    try {
      // 生成文件名
      const fileName = `recording-${Date.now()}.webm`;
      const filePath = `${userId}/${sessionId}/${fileName}`;

      // 上传到 Supabase Storage
      const { data, error } = await supabase.storage
        .from('live-recordings')
        .upload(filePath, recordedBlob, {
          contentType: 'video/webm',
        });

      if (error) throw error;

      // 获取公开 URL
      const { data: urlData } = supabase.storage
        .from('live-recordings')
        .getPublicUrl(filePath);

      // 保存记录到数据库
      const { data: recording, error: dbError } = await supabase
        .from('live_recordings')
        .insert({
          session_id: sessionId,
          title: `录制 - ${new Date().toLocaleString()}`,
          file_path: filePath,
          file_url: urlData.publicUrl,
          file_size: recordedBlob.size,
          recorded_by: userId,
          status: 'completed',
        })
        .select()
        .single();

      if (dbError) throw dbError;

      return recording;
    } catch (error) {
      console.error('上传录制失败:', error);
      throw error;
    }
  }, [recordedBlob]);

  return {
    isRecording,
    recordedBlob,
    startRecording,
    stopRecording,
    uploadRecording,
  };
}
```

## 📺 回放页面实现

### 创建回放列表页面

`src/app/recordings/page.tsx`：

```typescript
"use client";

import { useEffect, useState } from 'react';
import Link from 'next/link';
import { supabase } from '@/lib/supabase';

interface Recording {
  id: string;
  title: string;
  created_at: string;
  duration_seconds: number;
  file_size: number;
  status: string;
}

export default function RecordingsPage() {
  const [recordings, setRecordings] = useState<Recording[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function loadRecordings() {
      try {
        const { data, error } = await supabase
          .from('live_recordings')
          .select('*')
          .eq('status', 'completed')
          .order('created_at', { ascending: false });

        if (error) throw error;
        setRecordings(data || []);
      } catch (error) {
        console.error('加载录制列表失败:', error);
      } finally {
        setLoading(false);
      }
    }

    loadRecordings();
  }, []);

  const formatDuration = (seconds: number) => {
    const hours = Math.floor(seconds / 3600);
    const minutes = Math.floor((seconds % 3600) / 60);
    const secs = seconds % 60;
    return `${hours}:${minutes.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
  };

  const formatFileSize = (bytes: number) => {
    return (bytes / (1024 * 1024)).toFixed(2) + ' MB';
  };

  if (loading) {
    return <div className="p-8">加载中...</div>;
  }

  return (
    <div className="container mx-auto p-8">
      <h1 className="text-3xl font-bold mb-8">直播回放</h1>

      {recordings.length === 0 ? (
        <p className="text-gray-500">暂无录制</p>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
          {recordings.map((recording) => (
            <Link
              key={recording.id}
              href={`/recordings/${recording.id}`}
              className="bg-white rounded-lg shadow-md hover:shadow-lg transition-shadow p-6"
            >
              <h3 className="text-xl font-semibold mb-2">{recording.title}</h3>
              <div className="text-sm text-gray-600 space-y-1">
                <p>时长: {formatDuration(recording.duration_seconds)}</p>
                <p>大小: {formatFileSize(recording.file_size)}</p>
                <p>日期: {new Date(recording.created_at).toLocaleString()}</p>
              </div>
            </Link>
          ))}
        </div>
      )}
    </div>
  );
}
```

```typescript
"use client";

import { useEffect, useState } from 'react';
import { useParams } from 'next/navigation';
import { supabase } from '@/lib/supabase';

interface Recording {
  id: string;
  title: string;
  description: string;
  file_url: string;
  created_at: string;
  duration_seconds: number;
}

export default function RecordingPlayback() {
  const params = useParams();
  const [recording, setRecording] = useState<Recording | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function loadRecording() {
      try {
        const { data, error } = await supabase
          .from('live_recordings')
          .select('*')
          .eq('id', params.id)
          .single();

        if (error) throw error;
        setRecording(data);
      } catch (error) {
        console.error('加载录制失败:', error);
      } finally {
        setLoading(false);
      }

    if (params.id) {
      loadRecording();
    }
  }, [params.id]);

  if (loading) {
    return <div className="p-8">加载中...</div>;
  }

  if (!recording) {
    return <div className="p-8">录制不存在</div>;
  }

  return (

      {recording.description && (
        <p className="text-gray-600 mb-6">{recording.description}</p>
      )}

      <div className="bg-black rounded-lg overflow-hidden">
        <video
          controls
          className="w-full"
          src={recording.file_url}
          preload="metadata"
        >
          您的浏览器不支持视频播放。
        </video>
      </div>

      <div className="mt-6 text-sm text-gray-600">
        <p>录制日期: {new Date(recording.created_at).toLocaleString()}</p>
```
