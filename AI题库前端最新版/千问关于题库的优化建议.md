二、各环节实现方式与成本分析
1️⃣ OCR 识别：使用 DeepSeek-OCR
✅ 实现方式：
输入：扫描版 PDF 或图像（如 JPG/PNG）

输出：整页文本（含公式位置信息，支持 LaTeX）

关键优势：用视觉 token 压缩长文档，大幅降低 token 消耗 

例如：一页试卷 ≈ 300 文本 tokens → 压缩为 30 视觉 tokens

💰 成本计算（按 DeepSeek-V3 定价）：
输入定价（缓存命中）：¥0.5 / 百万 tokens 

输出定价：¥8 / 百万 tokens

假设：
每道题平均占 0.5 页

1000 题 ≈ 500 页

每页压缩后消耗 50 视觉 tokens（输入） + 300 文本 tokens（输出）

📌 注：DeepSeek-OCR 的输出是还原后的文本，需按文本 token 计费

总 token 消耗：
输入：500 页 × 50 = 25,000 tokens

输出：500 页 × 300 = 150,000 tokens

OCR 成本：
输入费用：25,000 ÷ 1e6 × ¥0.5 ≈ ¥0.0125

输出费用：150,000 ÷ 1e6 × ¥8 ≈ ¥1.20

合计 ≈ ¥1.21

✅ 结论：OCR 成本极低，可忽略不计






2️⃣ 题目切分：规则 + LLM 后处理
✅ 实现方式：
初步切分：用正则匹配题号（如 “1.”、“2)”、“【例题】”）

精细化切分：对模糊边界，调用 轻量 LLM（如 DeepSeek-V3） 辅助判断

但为控制成本，建议纯规则+启发式切分（初中几何题格式较规范），无需调用模型。
💡 若必须用模型辅助切分（如手写混排），每题额外约 200 tokens，成本增加 ¥0.16/题 → 不推荐。






3️⃣ 数学理解与打标签：DeepSeek-Math API
这是主要成本来源。
✅ Prompt 设计（结构化输出）：
你是一位初中数学教研专家，请分析以下几何题：

【题目】{ocr_text}

请严格按以下 JSON 格式输出，不要任何解释：
{
  "subject": "math",
  "grade": "junior_high",
  "topic": "geometry",
  "knowledge_points": ["全等三角形", "角平分线性质"],
  "difficulty": "medium",
  "question_type": "proof",
  "tags": ["中考模拟", "辅助线构造"]
}
💰 成本估算：
每题输入（题目）≈ 200 tokens

每题输出（JSON）≈ 100 tokens

DeepSeek-Math 调用走 DeepSeek-R1/V3 通用推理 API（无独立定价），按 R1 定价计算： 

输入：¥1 / 百万 tokens（缓存命中）

输出：¥16 / 百万 tokens

总消耗（1000 题）：
输入：1000 × 200 = 200,000 tokens

输出：1000 × 100 = 100,000 tokens

成本：
输入：200,000 ÷ 1e6 × ¥1 = ¥0.20

输出：100,000 ÷ 1e6 × ¥16 = ¥1.60

合计 ≈ ¥1.80

⚠️ 注意：若未命中缓存（首次调用），输入价格可能为 ¥2–5/百万，但影响有限。

    // 2. 根据内容类型和成本限制选择提供者
    let provider;
    if (contentProfile.hasMath && options.forceQuality) {
      provider = 'qwen-vl'; // 仅当有数学公式且强制高质量时使用
    } else if (options.maxCost && options.maxCost < 0.01) {
      provider = 'paddle-ocr'; // 严格成本控制时使用开源
    } else {
      provider = contentProfile.complexity > 0.7 ? 'qwen-vl' : 'paddle-ocr';
    }
    
    // 3. 执行解析
    console.log(`[COST] Using ${provider} for ${file.name}, estimated cost: ¥${this.estimateCost(provider, contentProfile)}`);
    return this.executeWithProvider(provider, file);
  }
  
  private static estimateCost(provider: string, profile: any) {
    const costMap = {
      'paddle-ocr': 0.002 * profile.pageCount,
      'qwen-vl': 0.02 * profile.pageCount
    };
    return costMap[provider] || 0;
  }
}

// app/api/ingest/upload/route.js - 超精简BFF
import { createClient } from '@supabase/supabase-js';
import { PaddleOCRProcessor } from '@/lib/ocr/paddle-ocr';

export async function POST(req) {
  const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_KEY);
  const { file, fileName, userId } = await req.json();
  
  try {
    // 1. 创建任务
    const { data: task, error: taskError } = await supabase
      .from('upload_tasks')
      .insert({ 
        user_id: userId, 
        file_name: fileName,
        file_url: 'temp', // 后面更新
        status: 'processing'
      })
      .select()
      .single();
    
    if (taskError) throw taskError;
    
    // 2. 上传文件到存储
    const fileExt = fileName.split('.').pop();
    const filePath = `uploads/${task.id}.${fileExt}`;
    const { error: storageError } = await supabase.storage
      .from('question-files')
      .upload(filePath, Buffer.from(file, 'base64'), { 
        contentType: `application/${fileExt === 'pdf' ? 'pdf' : 'msword'}` 
      });
    
    if (storageError) throw storageError;
    
    // 3. 更新文件URL
    const publicUrl = supabase.storage.from('question-files').getPublicUrl(filePath).data.publicUrl;
    await supabase
      .from('upload_tasks')
      .update({ file_url: publicUrl })
      .eq('id', task.id);
    
    // 4. 处理文件 (使用开源PaddleOCR)
    const processor = new PaddleOCRProcessor();
    const questions = await processor.parseFile(publicUrl);
    
    // 5. 保存解析结果
    for (const question of questions) {
      await supabase.from('parsed_questions').insert({
        upload_task_id: task.id,
        content: question.content,
        options: question.options || null,
        answer: question.answer || null,
        explanation: question.explanation || null,
        tags: JSON.stringify(question.tags || [])
      });
    }
    
    // 6. 更新任务状态
    await supabase
      .from('upload_tasks')
      .update({ status: 'completed' })
      .eq('id', task.id);
    
    return Response.json({ success: true, taskId: task.id, questionsCount: questions.length });
  } catch (error) {
    console.error('Processing error:', error);
    await supabase
      .from('upload_tasks')
      .update({ status: 'failed', error: error.message.substring(0, 255) })
      .eq('id', task?.id || '');
    return Response.json({ success: false, error: '处理失败' }, { status: 500 });
  }
}

function fileToBase64(file: File): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.readAsDataURL(file);
    reader.onload = () => resolve(reader.result?.toString().split(',')[1] || '');
    reader.onerror = reject;
  });
}

export class CostMonitor {
  static async checkQuota(userId: string) {
    const today = new Date().toISOString().split('T')[0];
    const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_KEY);
    
    // 获取今日用量
    const { data: usage, error } = await supabase
      .from('daily_usage')
      .select('api_calls, cost')
      .eq('user_id', userId)
      .eq('date', today)
      .single();
    
    // 独立开发者预算：¥10/天
    const dailyBudget = 10;
    const usedBudget = usage?.cost || 0;
    const remainingBudget = dailyBudget - usedBudget;
    
    return {
      remainingBudget,
      overBudget: remainingBudget < 0,
      usage: {
        apiCalls: usage?.api_calls || 0,
        cost: usedBudget
      }
    };
  }
  
  static async recordUsage(userId: string, cost: number, provider: string) {
    const today = new Date().toISOString().split('T')[0];
    const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_KEY);
    
    // 更新或创建今日用量记录
    const { error } = await supabase.rpc('upsert_daily_usage', {
      p_user_id: userId,
      p_date: today,
      p_cost: cost,
      p_api_calls: 1,
      p_providers: [provider]
    });
    
    if (error) console.error('Failed to record usage:', error);
  }
  
  static getCostEstimate(fileType: string, pageCount: number) {
    // 开源模型成本估算
    const costPerPage = {
      'standard': 0.002,   // PaddleOCR
      'math': 0.015,       // Qwen-VL (仅10%题目)
      'handwritten': 0.02  // Qwen-VL
    };
    
    return pageCount * (costPerPage[fileType] || costPerPage.standard);
  }
}
